cmake_minimum_required(VERSION 3.16)
project(QQBotHost VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MHD REQUIRED IMPORTED_TARGET libmicrohttpd)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.86.0)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(CJSON REQUIRED IMPORTED_TARGET libcjson)
pkg_check_modules(OPENSSL REQUIRED IMPORTED_TARGET openssl)
find_package(Threads REQUIRED)
add_library(qqbot_sdk SHARED sdk/src/qqbot_sdk.c)
target_include_directories(qqbot_sdk PUBLIC sdk/include)
target_compile_definitions(qqbot_sdk PRIVATE QQBOT_SDK_BUILD)
target_compile_options(qqbot_sdk PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(qqbot_sdk PRIVATE PkgConfig::CURL)
if(MINGW)
  target_link_options(qqbot_sdk PRIVATE -Wl,--kill-at)
endif()
set_target_properties(qqbot_sdk PROPERTIES OUTPUT_NAME qqbot_sdk VERSION 0.1.0 SOVERSION 0)

add_executable(qqbot src/main.c src/qq/webhook.c src/qq/events.c src/qq/event_queue.c src/qq/callback_security.c src/qq/gateway.c)
target_include_directories(qqbot PRIVATE src)
if(NOT WIN32)
  target_compile_definitions(qqbot PRIVATE _POSIX_C_SOURCE=200809L)
endif()
target_compile_options(qqbot PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(qqbot PRIVATE PkgConfig::MHD PkgConfig::CURL PkgConfig::SQLITE PkgConfig::CJSON PkgConfig::OPENSSL Threads::Threads)
if(WIN32)
  target_link_libraries(qqbot PRIVATE ws2_32)
endif()

enable_testing()
add_executable(test_event_parser tests/test_event_parser.c src/qq/events.c)
target_compile_options(test_event_parser PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_event_parser PRIVATE PkgConfig::CJSON)
add_test(NAME qq_group_event_parser COMMAND test_event_parser)

add_executable(test_event_queue tests/test_event_queue.c src/qq/event_queue.c)
target_compile_options(test_event_queue PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_event_queue PRIVATE PkgConfig::SQLITE)
add_test(NAME qq_event_queue COMMAND test_event_queue)

install(TARGETS qqbot RUNTIME DESTINATION bin)
install(TARGETS qqbot_sdk LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(FILES sdk/include/qqbot_sdk.h DESTINATION include)

add_executable(test_callback_security tests/test_callback_security.c src/qq/callback_security.c)
target_compile_options(test_callback_security PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(test_callback_security PRIVATE PkgConfig::CURL)
if(WIN32)
  target_link_libraries(test_callback_security PRIVATE ws2_32)
endif()
add_test(NAME qq_callback_security COMMAND test_callback_security)
